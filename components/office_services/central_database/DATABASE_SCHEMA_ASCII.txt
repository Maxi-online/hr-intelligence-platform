   └─► metadata, parameters, tags, dimensions, config_snapshot_*

1. ЗАГРУЗКА КОНФИГУРАЦИИ АГЕНТА:
   SELECT a.*, ap.system_prompt, ags.model, ags.reasoning_effort
   FROM agents a
   JOIN agent_prompts ap ON a.agent_id = ap.agent_id AND ap.is_current = TRUE
   JOIN agent_gpt_settings ags ON a.agent_id = ags.agent_id
   WHERE a.agent_id = 'supervisor_001';

2. ПОЛУЧЕНИЕ ИНСТРУМЕНТОВ АГЕНТА:
   SELECT at.tool_name, at.description, at.parameters
   FROM agent_tools_mapping atm
   JOIN agent_tools at ON atm.tool_id = at.tool_id
   WHERE atm.agent_id = 'recruiter_001' AND atm.is_enabled = TRUE
   ORDER BY atm.priority;

3. ЗАГРУЗКА CORE ПАМЯТИ:
   SELECT memory_key, memory_value, priority
   FROM core_memory
   WHERE agent_id = 'supervisor_001' AND is_active = TRUE
   ORDER BY priority;

4. ЛОГИРОВАНИЕ СООБЩЕНИЯ:
   INSERT INTO chat_history (message_id, agent_id, session_id, user_id, role, content)
   VALUES (UUID(), 'recruiter_001', 'sess_123', 'user_456', 'assistant', 'Ответ...');

5. ПОЛУЧЕНИЕ МАРШРУТОВ:
   SELECT s.agent_name AS from, t.agent_name AS to, r.routing_type
   FROM agent_routing_rules r
   JOIN agents s ON r.source_agent_id = s.agent_id
   JOIN agents t ON r.target_agent_id = t.agent_id
   WHERE r.is_active = TRUE;
